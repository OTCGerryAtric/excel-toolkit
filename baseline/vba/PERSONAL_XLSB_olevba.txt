Option Explicit

Public Sub ReportExternalLinks()
Attribute ReportExternalLinks.VB_ProcData.VB_Invoke_Func = "X\n14"

    Dim targetWorkbook As Workbook
    Dim reportSheet As Worksheet
    Dim worksheetItem As Worksheet
    Dim formulaCells As Range
    Dim cellItem As Range
    Dim nameItem As Name
    Dim chartItem As ChartObject
    Dim chartSheetItem As Chart
    Dim seriesItem As Series
    Dim linkSources As Variant
    Dim linkItem As Variant
    Dim nextRow As Long
    Dim resultsFound As Long

    Set targetWorkbook = ActiveWorkbook

    If targetWorkbook Is Nothing Then
        MsgBox "No workbook is currently active.", vbExclamation
        Exit Sub
    End If

    Application.ScreenUpdating = False
    Application.EnableEvents = False

    On Error GoTo ErrorHandler

    'Create or clear the report sheet.
    On Error Resume Next
    Set reportSheet = targetWorkbook.Worksheets("External Links Report")
    On Error GoTo ErrorHandler

    If reportSheet Is Nothing Then
        Set reportSheet = targetWorkbook.Worksheets.Add( _
            After:=targetWorkbook.Worksheets(targetWorkbook.Worksheets.Count))
        reportSheet.Name = "External Links Report"
    Else
        reportSheet.Cells.Clear
    End If

    reportSheet.Range("A1:E1").Value = Array( _
        "Location Type", _
        "Sheet / Scope", _
        "Location", _
        "External Link", _
        "Formula / Definition")

    nextRow = 2

    'Excel's registered external workbook links.
    linkSources = targetWorkbook.linkSources(Type:=xlLinkTypeExcelLinks)

    If IsArray(linkSources) Then
        For Each linkItem In linkSources
            AddExternalLinkResult _
                reportSheet, _
                nextRow, _
                "Registered Link", _
                "Workbook", _
                "", _
                CStr(linkItem), _
                ""

            resultsFound = resultsFound + 1
        Next linkItem
    End If

    'External references in cell formulas.
    For Each worksheetItem In targetWorkbook.Worksheets

        If worksheetItem.Name <> reportSheet.Name Then

            Set formulaCells = Nothing

            On Error Resume Next
            Set formulaCells = worksheetItem.UsedRange.SpecialCells( _
                xlCellTypeFormulas)
            On Error GoTo ErrorHandler

            If Not formulaCells Is Nothing Then
                For Each cellItem In formulaCells.Cells
                    If IsExternalReference(CStr(cellItem.Formula)) Then
                        AddExternalLinkResult _
                            reportSheet, _
                            nextRow, _
                            "Cell Formula", _
                            worksheetItem.Name, _
                            cellItem.Address(False, False), _
                            ExtractExternalWorkbook(CStr(cellItem.Formula)), _
                            CStr(cellItem.Formula)

                        resultsFound = resultsFound + 1
                    End If
                Next cellItem
            End If

        End If

    Next worksheetItem

    'Workbook-level and worksheet-level named ranges.
    For Each nameItem In targetWorkbook.Names

        If IsExternalReference(CStr(nameItem.RefersTo)) Then
            AddExternalLinkResult _
                reportSheet, _
                nextRow, _
                "Named Range", _
                GetNameScope(nameItem), _
                nameItem.Name, _
                ExtractExternalWorkbook(CStr(nameItem.RefersTo)), _
                CStr(nameItem.RefersTo)

            resultsFound = resultsFound + 1
        End If

    Next nameItem

    'External references in embedded charts.
    For Each worksheetItem In targetWorkbook.Worksheets

        If worksheetItem.Name <> reportSheet.Name Then

            For Each chartItem In worksheetItem.ChartObjects

                For Each seriesItem In chartItem.Chart.SeriesCollection

                    If IsExternalReference(CStr(seriesItem.Formula)) Then
                        AddExternalLinkResult _
                            reportSheet, _
                            nextRow, _
                            "Chart Series", _
                            worksheetItem.Name, _
                            chartItem.Name, _
                            ExtractExternalWorkbook( _
                                CStr(seriesItem.Formula)), _
                            CStr(seriesItem.Formula)

                        resultsFound = resultsFound + 1
                    End If

                Next seriesItem

            Next chartItem

        End If

    Next worksheetItem

    'External references in standalone chart sheets.
    For Each chartSheetItem In targetWorkbook.Charts

        For Each seriesItem In chartSheetItem.SeriesCollection

            If IsExternalReference(CStr(seriesItem.Formula)) Then
                AddExternalLinkResult _
                    reportSheet, _
                    nextRow, _
                    "Chart Series", _
                    chartSheetItem.Name, _
                    seriesItem.Name, _
                    ExtractExternalWorkbook(CStr(seriesItem.Formula)), _
                    CStr(seriesItem.Formula)

                resultsFound = resultsFound + 1
            End If

        Next seriesItem

    Next chartSheetItem

    'Format the report.
    With reportSheet

        .Rows(1).Font.Bold = True
        .Rows(1).Interior.Color = RGB(217, 225, 242)
        .Range("A1:E1").AutoFilter

        .Columns("A:D").AutoFit
        .Columns("E").ColumnWidth = 70
        .Columns("E").WrapText = True

        .Range("A1:E" & Application.Max(nextRow - 1, 1)).VerticalAlignment = _
            xlTop

        .Activate
        .Range("A2").Select

    End With

    Application.ScreenUpdating = True
    Application.EnableEvents = True

    MsgBox resultsFound & " external-link references found.", vbInformation
    Exit Sub

ErrorHandler:

    Application.ScreenUpdating = True
    Application.EnableEvents = True

    MsgBox "The external-links report could not be completed." & vbCrLf & _
           vbCrLf & _
           "Error " & Err.Number & ": " & Err.Description, _
           vbExclamation

End Sub

Private Function IsExternalReference( _
    ByVal formulaText As String) As Boolean

    Dim regularExpression As Object

    Set regularExpression = CreateObject("VBScript.RegExp")

    With regularExpression
        .Pattern = "\[[^\]]+\.(xlsx|xlsm|xlsb|xls|xlam|xla|csv)\]"
        .IgnoreCase = True
        .Global = False
    End With

    IsExternalReference = regularExpression.Test(formulaText)

End Function

Private Function ExtractExternalWorkbook( _
    ByVal formulaText As String) As String

    Dim regularExpression As Object
    Dim matches As Object

    Set regularExpression = CreateObject("VBScript.RegExp")

    With regularExpression
        .Pattern = "\[[^\]]+\.(xlsx|xlsm|xlsb|xls|xlam|xla|csv)\]"
        .IgnoreCase = True
        .Global = False
    End With

    Set matches = regularExpression.Execute(formulaText)

    If matches.Count > 0 Then
        ExtractExternalWorkbook = matches(0).Value
    Else
        ExtractExternalWorkbook = ""
    End If

End Function

Private Function GetNameScope(ByVal nameItem As Name) As String

    If TypeName(nameItem.Parent) = "Worksheet" Then
        GetNameScope = nameItem.Parent.Name
    Else
        GetNameScope = "Workbook"
    End If

End Function

Private Sub AddExternalLinkResult( _
    ByVal reportSheet As Worksheet, _
    ByRef nextRow As Long, _
    ByVal locationType As String, _
    ByVal scopeName As String, _
    ByVal locationName As String, _
    ByVal externalLink As String, _
    ByVal formulaText As String)

    reportSheet.Cells(nextRow, 1).Value = locationType
    reportSheet.Cells(nextRow, 2).Value = scopeName
    reportSheet.Cells(nextRow, 3).Value = locationName
    reportSheet.Cells(nextRow, 4).Value = externalLink
    reportSheet.Cells(nextRow, 5).Value = formulaText

    nextRow = nextRow + 1

End Sub
